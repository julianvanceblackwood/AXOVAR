cmake_minimum_required(VERSION 3.20)
project(AXOVAR VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(axovar_core core/main.cpp)
if(MSVC)
  target_compile_options(axovar_core PRIVATE /W4)
else()
  target_compile_options(axovar_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME engine_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_engine.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME api_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_api.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME intelligence_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_intelligence.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME audit_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_audit.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME portfolio_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_portfolio.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME settlement_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_settlement.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME g4_verification_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_g4.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME bounded_policy_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_bounded_model.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
add_test(NAME evidence_journal_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_evidence_journal.py" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")

# G5: same replay implementation, standalone in-process measurement binary.

  target_compile_options(axovar_bench PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

add_test(NAME g5_performance_contract_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_g5.py" "$<TARGET_FILE:axovar_bench>" "$<TARGET_FILE:axovar_core>" "${CMAKE_SOURCE_DIR}")
